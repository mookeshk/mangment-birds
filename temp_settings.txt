"use client";

import { useState, useEffect } from "react";
import { useAuth } from "../../../contexts/AuthContext";
import ConfirmModal from "../../../components/ConfirmModal";

const Icons = {
    Settings: () => <svg xmlns="http://www.w3.org/2000/svg" className="h-6 w-6" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M10.325 4.317c.426-1.756 2.924-1.756 3.35 0a1.724 1.724 0 002.573 1.066c1.543-.94 3.31.826 2.37 2.37a1.724 1.724 0 001.065 2.572c1.756.426 1.756 2.924 0 3.35a1.724 1.724 0 00-1.066 2.573c.94 1.543-.826 3.31-2.37 2.37a1.724 1.724 0 00-2.572 1.065c-.426 1.756-2.924 1.756-3.35 0a1.724 1.724 0 00-2.573-1.066c-1.543.94-3.31-.826-2.37-2.37a1.724 1.724 0 00-1.065-2.572c-1.756-.426-1.756-2.924 0-3.35a1.724 1.724 0 001.066-2.573c-.94-1.543.826-3.31 2.37-2.37.996.608 2.296.07 2.572-1.065z" /><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 12a3 3 0 11-6 0 3 3 0 016 0z" /></svg>,
    Plus: () => <svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M12 4v16m8-8H4" /></svg>,
    Trash: () => <svg xmlns="http://www.w3.org/2000/svg" className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-1.858L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" /></svg>,
    Edit: () => <svg xmlns="http://www.w3.org/2000/svg" className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15.232 5.232l3.536 3.536m-2.036-5.036a2.5 2.5 0 113.536 3.536L6.5 21.036H3v-3.572L16.732 3.732z" /></svg>,
    Close: () => <svg xmlns="http://www.w3.org/2000/svg" className="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" /></svg>
};

export default function SettingsPage() {
    const { fetchWithAuth } = useAuth();
    const [species, setSpecies] = useState<any[]>([]);
    const [breeds, setBreeds] = useState<any[]>([]);
    const [selectedSpeciesId, setSelectedSpeciesId] = useState<number | null>(null);
    const [activeTab, setActiveTab] = useState<'species' | 'password' | 'subscription'>('species');

    // Forms State
    const [newSpeciesName, setNewSpeciesName] = useState("");
    const [newMaturityAge, setNewMaturityAge] = useState<number | "">("");
    const [newIncubationPeriod, setNewIncubationPeriod] = useState<number | "">("");
    const [editSpeciesId, setEditSpeciesId] = useState<number | null>(null);

    const [newBreedName, setNewBreedName] = useState("");
    const [editBreedId, setEditBreedId] = useState<number | null>(null);

    const [deleteModal, setDeleteModal] = useState<{isOpen: boolean, type: 'species' | 'breed', id: number | null, title: string, message: string}>({
        isOpen: false,
        type: 'species',
        id: null,
        title: '',
        message: ''
    });

    useEffect(() => {
        loadSpecies();
    }, []);

    useEffect(() => {
        if (selectedSpeciesId) {
            loadBreeds(selectedSpeciesId);
        } else {
            setBreeds([]);
        }
    }, [selectedSpeciesId]);

    const loadSpecies = async () => {
        const res = await fetchWithAuth('/api/species');
        if (res.ok) {
            setSpecies(await res.json());
        }
    };

    const loadBreeds = async (speciesId: number) => {
        const res = await fetchWithAuth(`/api/breeds?speciesId=${speciesId}`);
        if (res.ok) {
            setBreeds(await res.json());
        }
    };

    // --- Species Actions ---

    const handleAddOrEditSpecies = async (e: React.FormEvent) => {
        e.preventDefault();
        if (!newSpeciesName.trim()) return;

        const url = editSpeciesId ? `/api/species/${editSpeciesId}` : '/api/species';
        const method = editSpeciesId ? 'PUT' : 'POST';

        const res = await fetchWithAuth(url, {
            method,
            body: JSON.stringify({ 
                name: newSpeciesName.trim(),
                maturityAgeInDays: newMaturityAge ? newMaturityAge * 30 : null,
                incubationPeriodInDays: newIncubationPeriod ? newIncubationPeriod : null
            })
        });

        if (res.ok) {
            resetSpeciesForm();
            loadSpecies();
        }
    };

    const startEditSpecies = (s: any) => {
        setEditSpeciesId(s.id);
        setNewSpeciesName(s.name);
        setNewMaturityAge(s.maturityAgeInDays ? Math.round(s.maturityAgeInDays / 30) : "");
        setNewIncubationPeriod(s.incubationPeriodInDays || "");
    };

    const resetSpeciesForm = () => {
        setEditSpeciesId(null);
        setNewSpeciesName("");
        setNewMaturityAge("");
        setNewIncubationPeriod("");
    };

    const confirmDeleteSpecies = (id: number) => {
        setDeleteModal({
            isOpen: true,
            type: 'species',
            id: id,
            title: 'حذف النوع',
            message: 'هل أنت متأكد من حذف هذا النوع؟'
        });
    };

    // --- Breed Actions ---

    const handleAddOrEditBreed = async (e: React.FormEvent) => {
        e.preventDefault();
        if (!newBreedName.trim() || !selectedSpeciesId) return;

        const url = editBreedId ? `/api/breeds/${editBreedId}` : '/api/breeds';
        const method = editBreedId ? 'PUT' : 'POST';

        const res = await fetchWithAuth(url, {
            method,
            body: JSON.stringify({ name: newBreedName.trim(), speciesId: selectedSpeciesId })
        });

        if (res.ok) {
            resetBreedForm();
            loadBreeds(selectedSpeciesId);
            loadSpecies();
        }
    };

    const startEditBreed = (b: any) => {
        setEditBreedId(b.id);
        setNewBreedName(b.name);
    };

    const resetBreedForm = () => {
        setEditBreedId(null);
        setNewBreedName("");
    };

    const confirmDeleteBreed = (id: number) => {
        setDeleteModal({
            isOpen: true,
            type: 'breed',
            id: id,
            title: 'حذف الفصيلة',
            message: 'هل أنت متأكد من حذف هذه الفصيلة؟'
        });
    };

    // --- Delete Execution ---

    const executeDelete = async () => {
        if (!deleteModal.id) return;

        if (deleteModal.type === 'species') {
            const res = await fetchWithAuth(`/api/species/${deleteModal.id}`, { method: 'DELETE' });
            if (res.ok) {
                if (selectedSpeciesId === deleteModal.id) setSelectedSpeciesId(null);
                if (editSpeciesId === deleteModal.id) resetSpeciesForm();
                loadSpecies();
            } else {
                const text = await res.text();
                alert(text || "لا يمكن حذف هذا النوع.");
            }
        } else if (deleteModal.type === 'breed') {
            const res = await fetchWithAuth(`/api/breeds/${deleteModal.id}`, { method: 'DELETE' });
            if (res.ok) {
                if (editBreedId === deleteModal.id) resetBreedForm();
                loadBreeds(selectedSpeciesId!);
                loadSpecies(); 
            } else {
                const text = await res.text();
                alert(text || "لا يمكن حذف هذه الفصيلة.");
            }
        }
        
        setDeleteModal({ ...deleteModal, isOpen: false });
    };

        return (
        <div className="p-4 sm:p-6 lg:p-8">
            <div className="max-w-6xl mx-auto space-y-6">

                {/* Header & Tabs */}
                <div className="bg-slate-900 rounded-3xl p-8 text-center relative overflow-hidden shadow-xl border border-slate-800">
                    <div className="absolute top-0 right-0 w-32 h-32 bg-emerald-500 rounded-full mix-blend-multiply filter blur-3xl opacity-20"></div>
                    <div className="absolute bottom-0 left-0 w-32 h-32 bg-blue-500 rounded-full mix-blend-multiply filter blur-3xl opacity-20"></div>
                    
                    <div className="relative z-10">
                        <div className="w-16 h-16 bg-slate-800 rounded-2xl flex items-center justify-center mx-auto mb-6 shadow-lg border border-slate-700 text-emerald-400">
                            <Icons.Settings />
                        </div>
                        <h1 className="text-3xl font-extrabold text-white mb-4">الإعدادات</h1>
                        
                        <div className="flex flex-wrap justify-center gap-2 mt-6">
                            <button onClick={() => setActiveTab('species')} className={`px-6 py-2 rounded-full font-medium transition-all ${activeTab === 'species' ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25' : 'bg-slate-800 text-slate-300 hover:bg-slate-700'}`}>الأنواع والفصائل</button>
                            <button onClick={() => setActiveTab('password')} className={`px-6 py-2 rounded-full font-medium transition-all ${activeTab === 'password' ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25' : 'bg-slate-800 text-slate-300 hover:bg-slate-700'}`}>كلمة المرور</button>
                            <button onClick={() => setActiveTab('subscription')} className={`px-6 py-2 rounded-full font-medium transition-all ${activeTab === 'subscription' ? 'bg-emerald-500 text-white shadow-lg shadow-emerald-500/25' : 'bg-slate-800 text-slate-300 hover:bg-slate-700'}`}>تفاصيل الاشتراك</button>
                        </div>
                    </div>
                </div>

                {/* Tab Content */}
                {activeTab === 'species' && (
                    <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
                        {/* Species Panel */}
                    <div className="bg-white dark:bg-slate-800 rounded-2xl shadow-sm border border-gray-100 dark:border-slate-700 flex flex-col h-[500px]">
                        <div className="p-5 border-b border-gray-100 dark:border-slate-700 flex justify-between items-center">
                            <h3 className="font-bold text-gray-800 dark:text-white text-lg">الأنواع (Species)</h3>
                            {editSpeciesId && (
                                <span className="text-xs font-medium bg-amber-100 text-amber-700 px-2 py-1 rounded-full dark:bg-amber-900/50 dark:text-amber-400">
                                    وضع التعديل
                                </span>
                            )}
                        </div>
                        <div className={`p-4 border-b border-gray-100 dark:border-slate-700 transition-colors ${editSpeciesId ? 'bg-amber-50/50 dark:bg-amber-900/10' : 'bg-gray-50/50 dark:bg-slate-900/50'}`}>
                            <form onSubmit={handleAddOrEditSpecies} className="flex flex-col gap-2">
                                <input
                                    type="text"
                                    required
                                    value={newSpeciesName}
                                    onChange={e => setNewSpeciesName(e.target.value)}
                                    placeholder="اسم النوع (مثال: حمام، دجاج...)"
                                    className="w-full px-4 py-2 bg-white dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl text-sm text-gray-900 dark:text-white placeholder-gray-400 dark:placeholder-gray-500 focus:outline-none focus:ring-2 focus:ring-emerald-500/50"
                                />
                                <div className="flex gap-2">
                                    <input
                                        type="number"
                                        value={newMaturityAge}
                                        onChange={e => setNewMaturityAge(e.target.value ? Number(e.target.value) : "")}
                                        placeholder="سن البلوغ (بالأشهر)"
                                        className="flex-1 px-4 py-2 bg-white dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl text-sm text-gray-900 dark:text-white placeholder-gray-400 dark:placeholder-gray-500 focus:outline-none focus:ring-2 focus:ring-emerald-500/50"
                                    />
                                    <input
                                        type="number"
                                        value={newIncubationPeriod}
                                        onChange={e => setNewIncubationPeriod(e.target.value ? Number(e.target.value) : "")}
                                        placeholder="فترة حضانة البيض (بالأيام)"
                                        className="flex-1 px-4 py-2 bg-white dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl text-sm text-gray-900 dark:text-white placeholder-gray-400 dark:placeholder-gray-500 focus:outline-none focus:ring-2 focus:ring-emerald-500/50"
                                    />
                                </div>
                                <div className="flex gap-2 mt-1">
                                    <button type="submit" className={`flex-1 text-white px-4 py-2 rounded-xl text-sm font-bold transition-all shadow-md ${editSpeciesId ? 'bg-amber-500 hover:bg-amber-600 shadow-amber-500/20' : 'bg-emerald-500 hover:bg-emerald-600 shadow-emerald-500/20'}`}>
                                        {editSpeciesId ? "حفظ التعديلات" : "إضافة النوع"}
                                    </button>
                                    {editSpeciesId && (
                                        <button type="button" onClick={resetSpeciesForm} className="px-4 py-2 bg-gray-200 hover:bg-gray-300 text-gray-700 dark:bg-slate-700 dark:hover:bg-slate-600 dark:text-gray-300 rounded-xl text-sm font-bold transition-all">
                                            إلغاء
                                        </button>
                                    )}
                                </div>
                            </form>
                        </div>
                        <div className="flex-1 overflow-y-auto p-4 space-y-2">
                            {species.length === 0 ? (
                                <div className="text-center text-gray-500 text-sm py-8">لا توجد أنواع مضافة حتى الآن.</div>
                            ) : (
                                species.map(s => (
                                    <div 
                                        key={s.id} 
                                        onClick={() => setSelectedSpeciesId(s.id)}
                                        className={`flex justify-between items-center p-3 rounded-xl cursor-pointer transition-colors border ${
                                            selectedSpeciesId === s.id 
                                            ? 'bg-emerald-50 border-emerald-200 dark:bg-emerald-500/10 dark:border-emerald-500/30' 
                                            : 'bg-white dark:bg-slate-800 border-gray-100 dark:border-slate-700 hover:border-emerald-200 dark:hover:border-slate-600'
                                        }`}
                                    >
                                        <div className="flex items-center gap-3">
                                            <div className="w-8 h-8 rounded-full bg-emerald-100 dark:bg-emerald-900/50 text-emerald-600 dark:text-emerald-400 flex items-center justify-center font-bold text-xs">
                                                {s.name.charAt(0)}
                                            </div>
                                            <div>
                                                <div className={`font-bold text-sm ${selectedSpeciesId === s.id ? 'text-emerald-700 dark:text-emerald-400' : 'text-gray-800 dark:text-gray-200'}`}>
                                                    {s.name}
                                                </div>
                                                <div className="text-xs text-gray-500 flex flex-wrap gap-x-2 gap-y-1 mt-1">
                                                    <span>{s.breedsCount} فصائل</span>
                                                    {s.maturityAgeInDays && <span>• بلوغ: {Math.round(s.maturityAgeInDays / 30)} شهر</span>}
                                                    {s.incubationPeriodInDays && <span>• حضانة: {s.incubationPeriodInDays} يوم</span>}
                                                </div>
                                            </div>
                                        </div>
                                        <div className="flex gap-1">
                                            <button 
                                                onClick={(e) => { e.stopPropagation(); startEditSpecies(s); }}
                                                className="p-2 text-blue-400 hover:bg-blue-50 hover:text-blue-600 dark:hover:bg-blue-900/20 rounded-lg transition-colors"
                                            >
                                                <Icons.Edit />
                                            </button>
                                            <button 
                                                onClick={(e) => { e.stopPropagation(); confirmDeleteSpecies(s.id); }}
                                                className="p-2 text-red-400 hover:bg-red-50 hover:text-red-600 dark:hover:bg-red-900/20 rounded-lg transition-colors"
                                            >
                                                <Icons.Trash />
                                            </button>
                                        </div>
                                    </div>
                                ))
                            )}
                        </div>
                    </div>

                    {/* Breeds Panel */}
                    <div className="bg-white dark:bg-slate-800 rounded-2xl shadow-sm border border-gray-100 dark:border-slate-700 flex flex-col h-[500px]">
                        <div className="p-5 border-b border-gray-100 dark:border-slate-700 flex justify-between items-center">
                            <h3 className="font-bold text-gray-800 dark:text-white text-lg">الفصائل (Breeds)</h3>
                            {selectedSpeciesId && (
                                <div className="flex items-center gap-2">
                                    {editBreedId && (
                                        <span className="text-xs font-medium bg-amber-100 text-amber-700 px-2 py-1 rounded-full dark:bg-amber-900/50 dark:text-amber-400">
                                            تعديل
                                        </span>
                                    )}
                                    <span className="text-xs font-medium bg-emerald-100 text-emerald-700 px-2 py-1 rounded-full dark:bg-emerald-900/50 dark:text-emerald-400">
                                        لنوع: {species.find(s => s.id === selectedSpeciesId)?.name}
                                    </span>
                                </div>
                            )}
                        </div>
                        
                        {!selectedSpeciesId ? (
                            <div className="flex-1 flex flex-col items-center justify-center text-gray-400 dark:text-gray-500 p-8 text-center space-y-4">
                                <Icons.Settings />
                                <p className="text-sm">يرجى اختيار نوع من القائمة لعرض وإدارة الفصائل التابعة له.</p>
                            </div>
                        ) : (
                            <>
                                <div className={`p-4 border-b border-gray-100 dark:border-slate-700 transition-colors ${editBreedId ? 'bg-amber-50/50 dark:bg-amber-900/10' : 'bg-gray-50/50 dark:bg-slate-900/50'}`}>
                                    <form onSubmit={handleAddOrEditBreed} className="flex gap-2">
                                        <input
                                            type="text"
                                            required
                                            value={newBreedName}
                                            onChange={e => setNewBreedName(e.target.value)}
                                            placeholder="مثال: زاجل، كش..."
                                            className="flex-1 px-4 py-2 bg-white dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl text-sm text-gray-900 dark:text-white placeholder-gray-400 dark:placeholder-gray-500 focus:outline-none focus:ring-2 focus:ring-emerald-500/50"
                                        />
                                        <button type="submit" className={`text-white px-4 py-2 rounded-xl text-sm font-bold transition-all shadow-md ${editBreedId ? 'bg-amber-500 hover:bg-amber-600 shadow-amber-500/20' : 'bg-emerald-500 hover:bg-emerald-600 shadow-emerald-500/20'}`}>
                                            {editBreedId ? 'حفظ' : 'إضافة'}
                                        </button>
                                        {editBreedId && (
                                            <button type="button" onClick={resetBreedForm} className="px-3 py-2 bg-gray-200 hover:bg-gray-300 text-gray-700 dark:bg-slate-700 dark:hover:bg-slate-600 dark:text-gray-300 rounded-xl transition-all">
                                                <Icons.Close />
                                            </button>
                                        )}
                                    </form>
                                </div>
                                <div className="flex-1 overflow-y-auto p-4 space-y-2">
                                    {breeds.length === 0 ? (
                                        <div className="text-center text-gray-500 text-sm py-8">لا توجد فصائل مسجلة لهذا النوع.</div>
                                    ) : (
                                        breeds.map(b => (
                                            <div key={b.id} className="flex justify-between items-center p-3 rounded-xl bg-gray-50 dark:bg-slate-900 border border-transparent hover:border-gray-200 dark:hover:border-slate-700 transition-colors group">
                                                <div className="font-semibold text-sm text-gray-700 dark:text-gray-300">
                                                    {b.name}
                                                </div>
                                                <div className="flex gap-1 opacity-100 sm:opacity-0 sm:group-hover:opacity-100 transition-opacity">
                                                    <button 
                                                        onClick={() => startEditBreed(b)}
                                                        className="p-1.5 text-blue-400 hover:bg-blue-50 hover:text-blue-600 dark:hover:bg-blue-900/20 rounded-lg transition-colors"
                                                    >
                                                        <Icons.Edit />
                                                    </button>
                                                    <button 
                                                        onClick={() => confirmDeleteBreed(b.id)}
                                                        className="p-1.5 text-red-400 hover:bg-red-50 hover:text-red-600 dark:hover:bg-red-900/20 rounded-lg transition-colors"
                                                    >
                                                        <Icons.Trash />
                                                    </button>
                                                </div>
                                            </div>
                                        ))
                                    )}
                                </div>
                            </>
                        )}
                    </div></div>)}

                {activeTab === 'password' && (
                    <div className="bg-white dark:bg-slate-800 p-8 rounded-2xl shadow-sm border border-gray-100 dark:border-slate-700 max-w-2xl mx-auto">
                        <h2 className="text-xl font-bold mb-6">تغيير كلمة المرور</h2>
                        <form className="space-y-4">
                            <div>
                                <label className="block text-sm mb-2">كلمة المرور الحالية</label>
                                <input type="password" placeholder="••••••••" className="w-full px-4 py-2 bg-gray-50 dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl" />
                            </div>
                            <div>
                                <label className="block text-sm mb-2">كلمة المرور الجديدة</label>
                                <input type="password" placeholder="••••••••" className="w-full px-4 py-2 bg-gray-50 dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl" />
                            </div>
                            <div>
                                <label className="block text-sm mb-2">تأكيد كلمة المرور الجديدة</label>
                                <input type="password" placeholder="••••••••" className="w-full px-4 py-2 bg-gray-50 dark:bg-slate-900 border border-gray-200 dark:border-slate-700 rounded-xl" />
                            </div>
                            <button type="button" className="bg-emerald-500 hover:bg-emerald-600 text-white px-6 py-2 rounded-xl font-bold mt-4">حفظ التغييرات</button>
                        </form>
                    </div>
                )}

                {activeTab === 'subscription' && (
                    <div className="bg-white dark:bg-slate-800 p-8 rounded-2xl shadow-sm border border-gray-100 dark:border-slate-700 max-w-2xl mx-auto text-center space-y-6">
                        <div className="w-20 h-20 bg-emerald-100 dark:bg-emerald-900/30 text-emerald-500 rounded-full flex items-center justify-center mx-auto">
                            <svg xmlns="http://www.w3.org/2000/svg" className="h-10 w-10" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z" /></svg>
                        </div>
                        <h2 className="text-2xl font-bold">اشتراكك نشط</h2>
                        <div className="bg-gray-50 dark:bg-slate-900 p-4 rounded-xl text-right inline-block min-w-full sm:min-w-[300px]">
                            <p className="text-sm text-gray-500 dark:text-gray-400 mb-1">نوع الباقة</p>
                            <p className="font-bold mb-4">الباقة الاحترافية (Pro)</p>
                            
                            <p className="text-sm text-gray-500 dark:text-gray-400 mb-1">تاريخ الانتهاء</p>
                            <p className="font-bold">2027/08/23</p>
                        </div>
                    </div>
                )}
            </div>

            <ConfirmModal 
                isOpen={deleteModal.isOpen}
                title={deleteModal.title}
                message={deleteModal.message}
                onConfirm={executeDelete}
                onCancel={() => setDeleteModal({ ...deleteModal, isOpen: false })}
            />
        </div>
    );
}
